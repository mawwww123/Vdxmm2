local ESPContainer = { Indicators = {} }

function ESPContainer:Add(target, options)
    if not target then return end
    local highlight = Instance.new("Highlight")
    highlight.Name = "ESP_" .. tostring(math.random(100000, 999999))
    highlight.Adornee = target
    highlight.FillTransparency = 0.7
    highlight.FillColor = options.AccentColor or Color3.new(1, 1, 0)
    highlight.OutlineColor = options.AccentColor or Color3.new(1, 1, 0)
    highlight.OutlineTransparency = 0
    highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    highlight.Parent = CoreGui

    local label = nil
    if options.ShowLabel then
        label = Instance.new("BillboardGui")
        label.Name = "Label_" .. tostring(math.random(100000, 999999))
        label.AlwaysOnTop = true
        label.MaxDistance = 99999
        label.Size = UDim2.new(0, 70, 0, 70)
        label.StudsOffset = Vector3.new(0, 2, 0)
        label.Adornee = target
        label.Parent = CoreGui

        local textLabel = Instance.new("TextLabel")
        textLabel.Size = UDim2.new(1, 0, 1, 0)
        textLabel.AnchorPoint = Vector2.new(0.5, 0.5)
        textLabel.Position = UDim2.new(0.5, 0, 0.5, 0)
        textLabel.BackgroundTransparency = 1
        textLabel.Font = Enum.Font.SourceSansBold
        textLabel.TextScaled = true
        textLabel.TextWrapped = true
        textLabel.TextColor3 = options.AccentColor or Color3.new(1, 1, 0)
        textLabel.Text = options.LabelText or "Target"
        textLabel.Parent = label
        Instance.new("UIStroke", textLabel)
    end

    self.Indicators[target] = { Highlight = highlight, Label = label, Options = options }
end

function ESPContainer:Remove(target)
    local data = self.Indicators[target]
    if not data then return end
    if data.Highlight then data.Highlight:Destroy() end
    if data.Label then data.Label:Destroy() end
    self.Indicators[target] = nil
end

function ESPContainer:RemoveGroup(groupName)
    for target, data in pairs(self.Indicators) do
        if data.Options.GroupName == groupName then
            self:Remove(target)
        end
    end
end

function ESPContainer:ClearAllGroups()
    for target, _ in pairs(self.Indicators) do
        self:Remove(target)
    end
end

local function findMurderer()
    for _, i in ipairs(Players:GetPlayers()) do
        if i.Backpack:FindFirstChild("Knife") then return i end
    end
    for _, i in ipairs(Players:GetPlayers()) do
        if i.Character and i.Character:FindFirstChild("Knife") then return i end
    end
    if playerData then
        for player, data in playerData do
            if data.Role == "Murderer" and Players:FindFirstChild(player) then
                return Players:FindFirstChild(player)
            end
        end
    end
    return nil
end

local function findSheriff()
    for _, i in ipairs(Players:GetPlayers()) do
        if i.Backpack:FindFirstChild("Gun") then return i end
    end
    for _, i in ipairs(Players:GetPlayers()) do
        if i.Character and i.Character:FindFirstChild("Gun") then return i end
    end
    if playerData then
        for player, data in playerData do
            if data.Role == "Sheriff" and Players:FindFirstChild(player) then
                return Players:FindFirstChild(player)
            end
        end
    end
    return nil
end

local function findSheriffThatsNotMe()
    for _, i in ipairs(Players:GetPlayers()) do
        if i == localplayer then continue end
        if i.Backpack:FindFirstChild("Gun") then return i end
    end
    for _, i in ipairs(Players:GetPlayers()) do
        if i == localplayer then continue end
        if i.Character and i.Character:FindFirstChild("Gun") then return i end
    end
    if playerData then
        for player, data in playerData do
            if data.Role == "Sheriff" and Players:FindFirstChild(player) then
                local p = Players:FindFirstChild(player)
                if p ~= localplayer then return p end
            end
        end
    end
    return nil
end

local function findNearestPlayer()
    local nearestPlayer, shortestDistance = nil, math.huge
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= localplayer and player.Character then
            local localRoot = localplayer.Character and localplayer.Character:FindFirstChild("HumanoidRootPart")
            local otherRoot = player.Character:FindFirstChild("HumanoidRootPart")
            if localRoot and otherRoot then
                local distance = (localRoot.Position - otherRoot.Position).Magnitude
                if distance < shortestDistance then
                    shortestDistance = distance
                    nearestPlayer = player
                end
            end
        end
    end
    return nearestPlayer
end

local function getMap()
    for _, o in ipairs(workspace:GetChildren()) do
        if o:FindFirstChild("CoinContainer") and o:FindFirstChild("Spawns") then
            return o
        end
    end
    return nil
end

local function getPredictedPosition(player, offset)
    if not player.Character then return Vector3.new(0,0,0) end
    local hrp = player.Character:FindFirstChild("UpperTorso") or player.Character:FindFirstChild("HumanoidRootPart")
    local hum = player.Character:FindFirstChild("Humanoid")
    if not hrp or not hum then return Vector3.new(0,0,0) end
    local velocity = hrp.AssemblyLinearVelocity
    local moveDir = hum.MoveDirection
    local predicted = hrp.Position + (velocity * Vector3.new(0.75, 0.5, 0.75)) * (offset / 15) + moveDir * offset
    predicted = predicted * (((localplayer:GetNetworkPing() * 1000) * ((offsetToPingMult - 1) * 0.01)) + 1)
    return predicted
end

local function reloadESP()
    if not playerESP then return end
    ESPContainer:RemoveGroup("players")
    for _, player in ipairs(Players:GetPlayers()) do
        if player == localplayer and hideMeEsp then continue end
        if player.Character then
            if player == findMurderer() then
                ESPContainer:Add(player.Character, {
                    AccentColor = Color3.new(1, 0, 0.0156863),
                    ShowLabel = true,
                    LabelText = "Murderer",
                    GroupName = "players"
                })
            elseif player == findSheriff() then
                ESPContainer:Add(player.Character, {
                    AccentColor = Color3.new(0, 0.6, 1),
                    ShowLabel = false,
                    GroupName = "players"
                })
            else
                ESPContainer:Add(player.Character, {
                    AccentColor = Color3.new(0, 1, 0.0313725),
                    ShowLabel = false,
                    GroupName = "players"
                })
            end
        end
    end
end

local FloatingButtons = {
    Buttons = {},
    GlobalLocked = false,
    ScreenGui = nil,
}

function FloatingButtons:_getGui()
    if self.ScreenGui and self.ScreenGui.Parent then
        return self.ScreenGui
    end
    local parent
    if gethui then
        local ok, hui = pcall(gethui)
        parent = ok and hui or CoreGui
    else
        parent = CoreGui
    end
    local sg = Instance.new("ScreenGui")
    sg.Name = "YARHM_MM2_Floating"
    sg.ResetOnSpawn = false
    sg.IgnoreGuiInset = true
    sg.DisplayOrder = 999
    sg.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    sg.Parent = parent
    self.ScreenGui = sg
    return sg
end

-- Apply unique shape to a frame
local function ApplyShape(obj, shape)
    -- Remove existing corner
    local existing = obj:FindFirstChildOfClass("UICorner")
    if existing then existing:Destroy() end

    local corner = Instance.new("UICorner")

    if shape == "circle" then
        corner.CornerRadius = UDim.new(1, 0)
    elseif shape == "diamond" then
        corner.CornerRadius = UDim.new(0, 8)
        obj.Rotation = 45
    elseif shape == "rounded" then
        corner.CornerRadius = UDim.new(0, 14)
    elseif shape == "hex" then
        -- Approximate hexagon with rounded corners
        corner.CornerRadius = UDim.new(0, 22)
    elseif shape == "squircle" then
        corner.CornerRadius = UDim.new(0.35, 0)
    else
        corner.CornerRadius = UDim.new(1, 0)
    end

    corner.Parent = obj
end

function FloatingButtons:Create(config)
    config = config or {}
    local name      = config.Name or ("Btn_" .. tostring(math.random(1000, 9999)))
    local iconName  = config.Icon or "lucide:circle"
    local shape     = config.Shape or "circle"
    local size      = config.Size or 52
    local position  = config.Position or UDim2.new(0, 120, 0.5, 0)
    local color     = config.Color or Color3.fromRGB(0, 170, 255)
    local callback  = config.Callback or function() end
    local onToggle  = config.OnToggle  -- optional: returns current state or nil
    local isToggle  = config.Toggle or false

    local sg = self:_getGui()

    -- Container (holds button + decorations, allows rotated shapes)
    local container = Instance.new("Frame")
    container.Name = "FB_" .. name
    container.Size = UDim2.fromOffset(size, size)
    container.Position = position
    container.AnchorPoint = Vector2.new(0.5, 0.5)
    container.BackgroundTransparency = 1
    container.BorderSizePixel = 0
    container.Parent = sg

    -- Glow ring (rotating dashed effect)
    local glowRing = Instance.new("ImageLabel")
    glowRing.Name = "GlowRing"
    glowRing.Size = UDim2.fromScale(1.35, 1.35)
    glowRing.Position = UDim2.fromScale(0.5, 0.5)
    glowRing.AnchorPoint = Vector2.new(0.5, 0.5)
    glowRing.BackgroundTransparency = 1
    glowRing.Image = "rbxassetid://1316045217" -- ring texture
    glowRing.ImageColor3 = color
    glowRing.ImageTransparency = 0.75
    glowRing.ZIndex = 1
    glowRing.Parent = container

    -- Pulse ring (expands and fades)
    local pulseRing = Instance.new("Frame")
    pulseRing.Name = "PulseRing"
    pulseRing.Size = UDim2.fromScale(1, 1)
    pulseRing.Position = UDim2.fromScale(0.5, 0.5)
    pulseRing.AnchorPoint = Vector2.new(0.5, 0.5)
    pulseRing.BackgroundTransparency = 1
    pulseRing.ZIndex = 2
    pulseRing.Parent = container
    local pulseCorner = Instance.new("UICorner")
    pulseCorner.CornerRadius = UDim.new(1, 0)
    pulseCorner.Parent = pulseRing
    local pulseStroke = Instance.new("UIStroke")
    pulseStroke.Color = color
    pulseStroke.Thickness = 2
    pulseStroke.Transparency = 1
    pulseStroke.Parent = pulseRing

    -- Main button (unique shape)
    local btn = Instance.new("TextButton")
    btn.Name = "Button"
    btn.Size = UDim2.fromScale(1, 1)
    btn.Position = UDim2.fromScale(0.5, 0.5)
    btn.AnchorPoint = Vector2.new(0.5, 0.5)
    btn.Text = ""
    btn.AutoButtonColor = false
    btn.BackgroundColor3 = Color3.fromRGB(20, 22, 27)
    btn.BackgroundTransparency = 0.1
    btn.BorderSizePixel = 0
    btn.ZIndex = 5
    btn.Parent = container
    ApplyShape(btn, shape)

    -- Reset rotation if diamond (icon stays upright)
    local innerRotation = (shape == "diamond") and -45 or 0

    -- Stroke
    local stroke = Instance.new("UIStroke")
    stroke.Color = color
    stroke.Thickness = 1.6
    stroke.Transparency = 0.2
    stroke.Parent = btn

    -- Inner glow frame (for decoration)
    local innerGlow = Instance.new("Frame")
    innerGlow.Name = "InnerGlow"
    innerGlow.Size = UDim2.fromScale(0.75, 0.75)
    innerGlow.Position = UDim2.fromScale(0.5, 0.5)
    innerGlow.AnchorPoint = Vector2.new(0.5, 0.5)
    innerGlow.BackgroundColor3 = color
    innerGlow.BackgroundTransparency = 0.88
    innerGlow.BorderSizePixel = 0
    innerGlow.ZIndex = 6
    innerGlow.Parent = btn
    ApplyShape(innerGlow, shape)

    -- Icon (from ModernV2 library)
    local icon = Instance.new("ImageLabel")
    icon.Name = "Icon"
    icon.Size = UDim2.fromScale(0.55, 0.55)
    icon.Position = UDim2.fromScale(0.5, 0.5)
    icon.AnchorPoint = Vector2.new(0.5, 0.5)
    icon.BackgroundTransparency = 1
    icon.Rotation = innerRotation
    icon.ImageColor3 = Color3.fromRGB(255, 255, 255)
    icon.ZIndex = 7
    icon.Parent = btn

    -- Use ModernV2 to resolve icon
    pcall(function()
        ModernV2:SetIconMode(icon, iconName)

    local lockBadge = Instance.new("ImageLabel")
    lockBadge.Name = "LockBadge"
    lockBadge.Size = UDim2.fromOffset(16, 16)
    lockBadge.Position = UDim2.new(1, 2, 0, -2)
    lockBadge.AnchorPoint = Vector2.new(1, 0)
    lockBadge.BackgroundColor3 = Color3.fromRGB(15, 17, 22)
    lockBadge.BorderSizePixel = 0
    lockBadge.ImageTransparency = 1
    lockBadge.BackgroundTransparency = 1
    lockBadge.ZIndex = 10
    lockBadge.Parent = container
    local lbCorner = Instance.new("UICorner")
    lbCorner.CornerRadius = UDim.new(1, 0)
    lbCorner.Parent = lockBadge
    pcall(function()
        ModernV2:SetIconMode(lockBadge, "lucide:lock")
    end)

    -- Toggle indicator (dot bottom-right)
    local toggleDot = Instance.new("Frame")
    toggleDot.Name = "ToggleDot"
    toggleDot.Size = UDim2.fromOffset(10, 10)
    toggleDot.Position = UDim2.new(1, -2, 1, -2)
    toggleDot.AnchorPoint = Vector2.new(1, 1)
    toggleDot.BackgroundColor3 = Color3.fromRGB(60, 60, 60)
    toggleDot.BorderSizePixel = 0
    toggleDot.Visible = isToggle
    toggleDot.ZIndex = 10
    toggleDot.Parent = container
    local tdCorner = Instance.new("UICorner")
    tdCorner.CornerRadius = UDim.new(1, 0)
    tdCorner.Parent = toggleDot
    local tdStroke = Instance.new("UIStroke")
    tdStroke.Color = Color3.fromRGB(255, 255, 255)
    tdStroke.Thickness = 1
    tdStroke.Transparency = 0.5
    tdStroke.Parent = toggleDot

    -- Button object wrapper
    local FB = {
        Name = name,
        Container = container,
        Button = btn,
        Icon = icon,
        GlowRing = glowRing,
        PulseRing = pulseRing,
        PulseStroke = pulseStroke,
        LockBadge = lockBadge,
        ToggleDot = toggleDot,
        Stroke = stroke,
        BaseColor = color,
        Shape = shape,
        Dragging = false,
        DragStart = nil,
        StartPos = nil,
        Dragged = false,
        Locked = false,
        ToggleState = false,
        IsToggle = isToggle,
        Callback = callback,
        OnToggle = onToggle,
        Connection = {},
    }

    -- =====================
    -- IDLE ANIMATION (icon pulse + ring rotation)
    -- =====================
    local idleTweens = {}
    local function startIdle()
        -- Glow ring rotation
        local ringSpin = TweenService:Create(
            glowRing,
            TweenInfo.new(4, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut, -1),
            { Rotation = 360 }
        )
        ringSpin:Play()
        table.insert(idleTweens, ringSpin)

        -- Icon subtle float (scale up/down)
        local iconPulse = TweenService:Create(
            icon,
            TweenInfo.new(1.4, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, -1, true),
            { Size = UDim2.fromScale(0.62, 0.62) }
        )
        iconPulse:Play()
        table.insert(idleTweens, iconPulse)

        -- Pulse ring expanding
        task.spawn(function()
            while FB.Container and FB.Container.Parent do
                pulseRing.Size = UDim2.fromScale(1, 1)
                pulseStroke.Transparency = 0.4
                TweenService:Create(
                    pulseRing,
                    TweenInfo.new(1.8, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                    { Size = UDim2.fromScale(1.8, 1.8) }
                ):Play()
                TweenService:Create(
                    pulseStroke,
                    TweenInfo.new(1.8, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                    { Transparency = 1 }
                ):Play()
                task.wait(2)
            end
        end)
    end
    startIdle()

    -- =====================
    -- HOVER ANIMATION
    -- =====================
    local function onHoverEnter()
        TweenService:Create(btn, TweenInfo.new(0.25, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
            Size = UDim2.fromScale(1.12, 1.12),
            BackgroundTransparency = 0,
        }):Play()
        TweenService:Create(stroke, TweenInfo.new(0.25), {
            Transparency = 0,
            Thickness = 2.4,
        }):Play()
        TweenService:Create(icon, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
            Size = UDim2.fromScale(0.7, 0.7),
            Rotation = innerRotation + 8,
        }):Play()
    end

    local function onHoverExit()
        TweenService:Create(btn, TweenInfo.new(0.25, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
            Size = UDim2.fromScale(1, 1),
            BackgroundTransparency = 0.1,
        }):Play()
        TweenService:Create(stroke, TweenInfo.new(0.25), {
            Transparency = 0.2,
            Thickness = 1.6,
        }):Play()
        TweenService:Create(icon, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
            Size = UDim2.fromScale(0.55, 0.55),
            Rotation = innerRotation,
        }):Play()
    end

    table.insert(FB.Connection, btn.MouseEnter:Connect(onHoverEnter))
    table.insert(FB.Connection, btn.MouseLeave:Connect(onHoverExit))

    -- =====================
    -- PRESS ANIMATION + RIPPLE
    -- =====================
    local function onPress()
        TweenService:Create(btn, TweenInfo.new(0.08), {
            Size = UDim2.fromScale(0.85, 0.85),
        }):Play()
        TweenService:Create(icon, TweenInfo.new(0.25, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
            Rotation = icon.Rotation + 360,
        }):Play()

        -- Ripple
        local ripple = Instance.new("Frame")
        ripple.Size = UDim2.fromScale(0.2, 0.2)
        ripple.Position = UDim2.fromScale(0.5, 0.5)
        ripple.AnchorPoint = Vector2.new(0.5, 0.5)
        ripple.BackgroundColor3 = color
        ripple.BackgroundTransparency = 0.4
        ripple.BorderSizePixel = 0
        ripple.ZIndex = 4
        ripple.Parent = btn
        local rc = Instance.new("UICorner")
        rc.CornerRadius = UDim.new(1, 0)
        rc.Parent = ripple

        TweenService:Create(ripple, TweenInfo.new(0.7, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Size = UDim2.fromScale(2.5, 2.5),
            BackgroundTransparency = 1,
        }):Play()
        task.delay(0.7, function()
            if ripple.Parent then ripple:Destroy() end
        end)
    end

    local function onRelease()
        TweenService:Create(btn, TweenInfo.new(0.2, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
            Size = UDim2.fromScale(1, 1),
        }):Play()
    end

    -- =====================
    -- DRAG SUPPORT (with threshold)
    -- =====================
    local DRAG_THRESHOLD = 5
    local function onInputBegan(input)
        if input.UserInputType ~= Enum.UserInputType.MouseButton1
           and input.UserInputType ~= Enum.UserInputType.Touch then return end
        FB.DragStart = input.Position
        FB.StartPos = container.Position
        FB.Dragged = false
        FB.Dragging = true

        local moveConn, endConn
        moveConn = UserInputService.InputChanged:Connect(function(mv)
            if not FB.Dragging then return end
            if mv.UserInputType ~= Enum.UserInputType.MouseMovement
               and mv.UserInputType ~= Enum.UserInputType.Touch then return end
            local delta = mv.Position - FB.DragStart
            if delta.Magnitude > DRAG_THRESHOLD and not FB.Locked and not FloatingButtons.GlobalLocked then
                FB.Dragged = true
                container.Position = UDim2.new(
                    FB.StartPos.X.Scale, FB.StartPos.X.Offset + delta.X,
                    FB.StartPos.Y.Scale, FB.StartPos.Y.Offset + delta.Y
                )
            end
        end)
        endConn = input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                FB.Dragging = false
                moveConn:Disconnect()
                endConn:Disconnect()
            end
        end)
    end

    table.insert(FB.Connection, btn.InputBegan:Connect(onInputBegan))
    table.insert(FB.Connection, btn.MouseButton1Down:Connect(onPress))
    table.insert(FB.Connection, btn.MouseButton1Up:Connect(onRelease))
    table.insert(FB.Connection, btn.MouseLeave:Connect(onRelease))

    -- =====================
    -- CLICK HANDLER
    -- =====================
    table.insert(FB.Connection, btn.MouseButton1Click:Connect(function()
        if FB.Dragged then
            FB.Dragged = false
            return
        end

        if FB.IsToggle then
            FB.ToggleState = not FB.ToggleState
            local newColor = FB.ToggleState and Color3.fromRGB(80, 220, 120) or FB.BaseColor
            TweenService:Create(stroke, TweenInfo.new(0.3), { Color = newColor }):Play()
            TweenService:Create(glowRing, TweenInfo.new(0.3), { ImageColor3 = newColor }):Play()
            TweenService:Create(pulseStroke, TweenInfo.new(0.3), { Color = newColor }):Play()
            TweenService:Create(toggleDot, TweenInfo.new(0.3), {
                BackgroundColor3 = FB.ToggleState and Color3.fromRGB(80, 220, 120) or Color3.fromRGB(60, 60, 60)
            }):Play()
            FB.Callback(FB.ToggleState, FB)
        else
            FB.Callback(FB)
        end
    end))

    -- =====================
    -- PUBLIC API
    -- =====================
    function FB:SetLocked(state)
        self.Locked = state
        lockBadge.ImageTransparency = state and 0 or 1
        lockBadge.BackgroundTransparency = state and 0.15 or 1
    end

    function FB:SetToggleState(state)
        self.ToggleState = state
        local newColor = state and Color3.fromRGB(80, 220, 120) or self.BaseColor
        TweenService:Create(stroke, TweenInfo.new(0.3), { Color = newColor }):Play()
        TweenService:Create(toggleDot, TweenInfo.new(0.3), {
            BackgroundColor3 = state and Color3.fromRGB(80, 220, 120) or Color3.fromRGB(60, 60, 60)
        }):Play()
    end

    function FB:SetVisible(state)
        container.Visible = state
    end

    function FB:Destroy()
        for _, conn in ipairs(self.Connection) do
            pcall(function() conn:Disconnect() end)
        end
        for _, tw in ipairs(idleTweens) do
            pcall(function() tw:Cancel() end)
        end
        container:Destroy()
    end

    table.insert(self.Buttons, FB)
    return FB
end

function FloatingButtons:SetAllLocked(state)
    self.GlobalLocked = state
    for _, fb in ipairs(self.Buttons) do
        fb:SetLocked(state)
    end
end

function FloatingButtons:ClearAll()
    for _, fb in ipairs(self.Buttons) do
        fb:Destroy()
    end
    self.Buttons = {}
end

local function doShootMurderer()
    if findSheriff() ~= localplayer then
        Notify("You're not sheriff/hero.", 3) return
    end
    local murderer = findMurderer() or findSheriffThatsNotMe()
    if not murderer then Notify("No murderer to shoot.", 3) return end
    if not localplayer.Character:FindFirstChild("Gun") then
        if localplayer.Backpack:FindFirstChild("Gun") then
            localplayer.Character.Humanoid:EquipTool(localplayer.Backpack:FindFirstChild("Gun"))
        else Notify("You don't have the gun..?", 3) return end
    end
    local murdererHRP = murderer.Character:FindFirstChild("HumanoidRootPart")
    if not murdererHRP then return end
    local predicted = getPredictedPosition(murderer, shootOffset)
    local args
    if instakillshoot then
        args = { CFrame.new(murdererHRP.Position + Vector3.new(0, 1, 0)), CFrame.new(murdererHRP.Position) }
    else
        args = { CFrame.new(localplayer.Character.RightHand.Position), CFrame.new(predicted) }
    end
    localplayer.Character:WaitForChild("Gun"):WaitForChild("Shoot"):FireServer(unpack(args))
end

local AntiFlingRight = Tabs.Misc:AddSection({ Name = "Anti-Fling V2", Position = "Right" })

local antiFling = false
local antiFlingConnections = {}
local antiFlingTrackedPlayers = {}

function antifling_SetupCharacterCollision(character)
    local function disableCollide(part)
        if antiFling and part:IsA("BasePart") then
            part.CanCollide = false
        end
    end

    for _, part in ipairs(character:GetChildren()) do
        disableCollide(part)
    end

    local childAddedConn = character.ChildAdded:Connect(disableCollide)
    local steppedConn = RunService.Stepped:Connect(function()
        if antiFling and character:IsDescendantOf(workspace) then
            for _, part in ipairs(character:GetChildren()) do
                if part:IsA("BasePart") and part.CanCollide then
                    part.CanCollide = false
                end
            end
        end
    end)

    character.Destroying:Connect(function()
        childAddedConn:Disconnect()
        steppedConn:Disconnect()
    end)

    table.insert(antiFlingConnections, childAddedConn)
    table.insert(antiFlingConnections, steppedConn)
end

function antifling_TrackPlayer(player)
    if player == localplayer then return end
    if antiFlingTrackedPlayers[player] then return end
    antiFlingTrackedPlayers[player] = true
    local conn = player.CharacterAdded:Connect(antifling_SetupCharacterCollision)
    if player.Character then
        antifling_SetupCharacterCollision(player.Character)
    end
    table.insert(antiFlingConnections, conn)
end

function antifling_Uninstall()
    for _, conn in ipairs(antiFlingConnections) do
        pcall(function() conn:Disconnect() end)
    end
    antiFlingConnections = {}
    antiFlingTrackedPlayers = {}
end

function antifling_Enable()
    antiFling = true
    for _, player in ipairs(Players:GetPlayers()) do
        antifling_TrackPlayer(player)
    end
end

function antifling_Disable()
    antiFling = false
    antifling_Uninstall()
end

Players.PlayerAdded:Connect(function(player)
    if antiFling then antifling_TrackPlayer(player) end
end)

-- Auto shoot loop
task.spawn(function()
    while task.wait(1) do
        if findSheriff() == localplayer and autoShooting then
            repeat
                task.wait(0.1)
                local murderer = findMurderer()
                if not murderer then continue end
                local murdererPos = murderer.Character.HumanoidRootPart.Position
                local charRoot = localplayer.Character.HumanoidRootPart
                local rayDir = murdererPos - charRoot.Position
                local params = RaycastParams.new()
                params.FilterType = Enum.RaycastFilterType.Exclude
                params.FilterDescendantsInstances = {localplayer.Character}
                local hit = workspace:Raycast(charRoot.Position, rayDir, params)
                if not hit or hit.Instance.Parent == murderer.Character then
                    if not localplayer.Character:FindFirstChild("Gun") then
                        if localplayer.Backpack:FindFirstChild("Gun") then
                            localplayer.Character.Humanoid:EquipTool(localplayer.Backpack:FindFirstChild("Gun"))
                        else break end
                    end
                    local murdererHRP = murderer.Character:FindFirstChild("HumanoidRootPart")
                    if not murdererHRP then break end
                    local predicted = getPredictedPosition(murderer, shootOffset)
                    local args = {[1] = 1, [2] = predicted, [3] = "AH2"}
                    localplayer.Character.Gun.KnifeLocal.CreateBeam.RemoteFunction:InvokeServer(unpack(args))
                end
            until findSheriff() ~= localplayer or not autoShooting
        end
    end
end)
